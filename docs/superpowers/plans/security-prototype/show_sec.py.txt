"""Show findings of one security-prototype rule in a fixed seeded order: python show_sec.py <rule> <start> <count> [context]."""
import random
import sys

DIRS = {
    "auth-framework": "auth-framework", "synapse": "synapse", "mlmf": "mlmf", "lightbulb": "lightbulb", "lsp-bridge": "lsp-bridge",
    "fuel": "../perf/fuel", "baracuda": "../perf/baracuda", "Unpopped": "../perf/Unpopped",
}
rule, start, count = sys.argv[1], int(sys.argv[2]), int(sys.argv[3])
ctx = int(sys.argv[4]) if len(sys.argv) > 4 else 5
rows = []
for repo in DIRS:
    for line in open(f"out_{repo}.tsv", encoding="utf-8"):
        p = line.rstrip("\n").split("\t")
        if len(p) >= 4 and p[1] == rule:
            rows.append(p)
rows.sort(key=lambda r: (r[0], r[2]))
random.Random(2026).shuffle(rows)
print(f"# {rule}: population {len(rows)}; showing {start + 1}..{min(start + count, len(rows))}")
for i, (repo, _, loc, _src) in enumerate(rows[start : start + count], start + 1):
    rel, ln = loc.rsplit(":", 1)
    ln = int(ln)
    text = open(f"{DIRS[repo]}/{rel}", encoding="utf-8", errors="replace").read().split("\n")
    print(f"===== #{i} {repo} {loc}")
    for n in range(max(0, ln - 1 - ctx), min(len(text), ln + ctx)):
        mark = ">>" if n + 1 == ln else "  "
        print(f"{mark}{n + 1:5d} {text[n][:130]}")
