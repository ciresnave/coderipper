ORDER = [
    "secret-field-not-zeroized", "debug-prints-secret-field", "secret-named-value-in-log", "non-constant-time-compare", "aead-without-aad",
    "aead-empty-aad", "weak-algorithm", "tls-verification-off", "hardcoded-key-material", "hardcoded-secret:const",
]
repos = ["auth-framework", "synapse", "mlmf", "lightbulb", "lsp-bridge", "fuel", "baracuda", "Unpopped"]
summary, out = [], []
for rule in ORDER:
    rows = []
    for repo in repos:
        for line in open(f"out_{repo}.tsv", encoding="utf-8"):
            p = line.rstrip("\n").split("\t")
            if len(p) >= 4 and p[1] == rule:
                rows.append(p)
    rows.sort(key=lambda r: (r[0], r[2]))
    random.Random(2026).shuffle(rows)
    judged = sorted(V[rule])
    assert len(judged) <= len(rows), (rule, len(judged), len(rows))
    pt = sum(1 for i in judged if V[rule][i][0] == Y)
    at = sum(1 for i in judged if V[rule][i][1] == Y)
    summary.append((rule, len(rows), len(judged), pt, at))
    out.append(f"\n#### `{rule}` ({len(rows)} findings; {len(judged)} judged; detector right {pt}, reviewer acts {at})\n")
    out.append("| # | site | right | acts | why |\n|---|---|---|---|---|")
    for i in judged:
        repo, _, loc, _ = rows[i - 1]
        p, a, why = V[rule][i]
        out.append(f"| {i} | {repo} `{loc}` | {p} | {a} | {why} |")
print("SUMMARY")
for rule, pop, n, pt, at in summary:
    print(f"{rule}|{pop}|{n}|{pt}|{at}|{100 * pt // n}%|{100 * at // n}%")
print("TABLES")
print("\n".join(out))
