out = []
summary = []
for rule in ORDER:
    rows = []
    for repo in ["fuel", "baracuda", "Unpopped"]:
        for line in open(f"out2_{repo}.tsv", encoding="utf-8"):
            p = line.rstrip("\n").split("\t")
            if len(p) >= 4 and p[1] == rule:
                rows.append(p)
    rows.sort(key=lambda r: (r[0], r[2]))
    random.Random(2026).shuffle(rows)
    judged = sorted(V[rule])
    tp = sum(1 for i in judged if V[rule][i][0] == T)
    summary.append((rule, len(rows), len(judged), tp))
    out.append(f"\n#### `{rule}` ({len(rows)} findings; {len(judged)} judged, {tp} true)\n")
    out.append("| # | site | verdict | why |\n|---|---|---|---|")
    for i in judged:
        repo, _, loc, _ = rows[i - 1]
        v, why = V[rule][i]
        out.append(f"| {i} | {repo} `{loc}` | {v} | {why} |")
print("SUMMARY")
for rule, pop, n, tp in summary:
    print(f"| `{rule}` | {pop} | {n} | {tp} | {100 * tp // n}% |")
