"""Show findings of one rule in a fixed seeded order: python show.py <rule> <start> <count> [context]. Order = random.Random(2026).shuffle over
the rule's findings sorted by (repo, file, line), so the same call always shows the same items."""
import random
import sys

rule, start, count = sys.argv[1], int(sys.argv[2]), int(sys.argv[3])
ctx = int(sys.argv[4]) if len(sys.argv) > 4 else 6
rows = []
for repo in ["fuel", "baracuda", "Unpopped"]:
    for line in open(f"out2_{repo}.tsv", encoding="utf-8"):
        p = line.rstrip("\n").split("\t")
        if len(p) >= 4 and p[1] == rule:
            rows.append(p)
rows.sort(key=lambda r: (r[0], r[2]))
random.Random(2026).shuffle(rows)
print(f"# {rule}: population {len(rows)}; showing {start + 1}..{min(start + count, len(rows))}")
for i, (repo, _, loc, _src) in enumerate(rows[start : start + count], start + 1):
    rel, ln = loc.rsplit(":", 1)
    ln = int(ln)
    text = open(f"{repo}/{rel}", encoding="utf-8", errors="replace").read().split("\n")
    print(f"===== #{i} {repo} {loc}")
    for n in range(max(0, ln - 1 - ctx), min(len(text), ln + ctx)):
        mark = ">>" if n + 1 == ln else "  "
        print(f"{mark}{n + 1:5d} {text[n][:140]}")
